if __name__ == '__main__':
    n = int(input())
    student_marks = {}
    for _ in range(n):
        name, *line = input().split()
        scores = list(map(float, line))
        student_marks[name] = scores
    query_name = input()
    for i in student_marks:
        if i==query_name:
            a=student_marks[i]
    s=a[0]+a[1]+a[2]
    avg=s/3
    print(f"{avg:.2f}")
